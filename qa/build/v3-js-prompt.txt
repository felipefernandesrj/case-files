function setLocView(view) {
  const loc = document.getElementById('locations');
  if (!loc) return;
  loc.dataset.loc = view;
  const timeBtn = document.getElementById('loc-time');
  const roomBtn = document.getElementById('loc-rooms');
  if (timeBtn) timeBtn.setAttribute('aria-pressed', view === 'time' ? 'true' : 'false');
  if (roomBtn) roomBtn.setAttribute('aria-pressed', view === 'rooms' ? 'true' : 'false');
}
function renderDossier() {
  const body = document.getElementById('dossier-body');
  if (!body || !caseData) return;
  body.innerHTML = '';
  const s = (caseData.suspects || []).find(x => x.id === selectedSuspect);
  if (!s) {
    body.textContent = lang === 'pt' ? 'Escolha um suspeito na lista.' : 'Choose a suspect from the list.';
    return;
  }
  const face = document.createElement('span');
  face.className = 'portrait';
  face.setAttribute('aria-hidden', 'true');
  face.textContent = initials(s.name);
  const who = document.createElement('strong');
  who.textContent = s.name;
  const role = document.createElement('p');
  role.className = 'role with-icon';
  const iconId = ROLE_ICONS[s.id];
  if (iconId) role.appendChild(iconSvg(iconId, 'icon badge-icon'));
  role.appendChild(document.createTextNode(s.role || ''));
  body.appendChild(face);
  body.appendChild(who);
  if (s.role) body.appendChild(role);
}
