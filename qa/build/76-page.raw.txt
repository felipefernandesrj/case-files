async function accuse(suspectId){
  const playerName = document.getElementById('player').value.trim()||'anon';
  try{
    const res = await fetch(API + '/accuse', {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({sessionId, suspectId, player:playerName, lang})
    });
    const data = await res.json();
    if (!res.ok) throw data;
    lastVerdict = data;
    renderVerdict(lastVerdict);
    await loadScores();
    await loadCredits();
  }catch(err){
    showError(err.message||(lang==='pt'?'Erro':'Error'));
  }
}
function renderVerdict(data) {
  const verdict = document.getElementById('verdict');
  verdict.hidden = !data;
  if (!data) return;
  verdict.innerHTML = `<p>${data.correct ? (lang==='pt'?'Você acertou!':'Correct!') : (lang==='pt'?'Você errou!':'Wrong!')}</p>
    <p>${lang==='pt'?'Culpado':'Culprit'}: ${data.culpritName}</p>
    <p>${lang==='pt'?'Solução':'Solution'}: ${data.solution}</p>
    <p>${lang==='pt'?'Pontuação':'Score'}: ${data.score}</p>`;
  verdict.focus();
}

async function loadScores(){
  const sec = document.getElementById('scores-section');
  const tbody = document.getElementById('scores-body');
  const res = await fetch(API + '/scores');
  const data = await res.json();
  tbody.innerHTML = '';
  data.forEach((row,i)=>{
    const tr = document.createElement('tr');
    tr.innerHTML = `<td>${i+1}</td><td>${row.player}</td><td>${row.score}</td><td>${row.questions}</td>`;
    tbody.appendChild(tr);
  });
  sec.hidden = false;
}

async function loadCredits() {
  try {
    const res = await fetch(API + '/credits?lang=' + lang);
    if (!res.ok) return;
    lastCredits = await res.json();
    renderCredits(lastCredits);
  } catch (e) {}
}
function renderCredits(data) {
  if (!data) return;
  const sec = document.getElementById('credits');
  sec.hidden = false;
  document.getElementById('credits-title').textContent = data.title;
  const list = document.getElementById('credits-list');
  list.innerHTML = '';
  data.backers.forEach(b => {
    const li = document.createElement('li');
    li.textContent = b;
    list.appendChild(li);
  });
  const note = document.getElementById('credits-note');
  if (data.example) {
    note.textContent = lang === 'pt' ? 'Nomes de exemplo' : 'Example names';
    note.hidden = false;
  } else {
    note.hidden = true;
  }
  sec.focus();
}
function showError(msg){
  const el = document.getElementById('err');
  el.textContent = msg;
  el.hidden = false;
  setTimeout(()=>{el.hidden=true;},5000);
}
