async function loadPackScores() {
    const title = document.getElementById('pack-scores-title');
    const caption = document.getElementById('pack-scores-caption');
    const thPlayer = document.getElementById('th-pack-player');
    const thSolved = document.getElementById('th-pack-solved');
    const thTotal = document.getElementById('th-pack-total');
    const costLink = document.getElementById('cost-link');
    const tbody = document.getElementById('pack-scores-body');
    const empty = document.getElementById('pack-scores-empty');

    if (title) title.textContent = UI[lang].packScores;
    if (caption) caption.textContent = UI[lang].packCaption;
    if (thPlayer) thPlayer.textContent = UI[lang].packPlayer;
    if (thSolved) thSolved.textContent = UI[lang].packSolved;
    if (thTotal) thTotal.textContent = UI[lang].packTotal;
    if (costLink) {
        costLink.textContent = UI[lang].costLink;
        costLink.href = '/costs?lang=' + lang;
    }

    try {
        const res = await fetch(`${API}/scores?case=all`);
        const scores = await res.json();
        tbody.innerHTML = '';
        empty.hidden = true;

        if (!scores || scores.length === 0) {
            throw new Error('Empty');
        }

        scores.forEach((s, i) => {
            const tr = document.createElement('tr');
            const tdRank = document.createElement('td');
            tdRank.textContent = i + 1;
            const tdPlayer = document.createElement('td');
            tdPlayer.textContent = s.player;
            const tdSolved = document.createElement('td');
            tdSolved.textContent = s.solved;
            const tdTotal = document.createElement('td');
            tdTotal.textContent = s.total;
            tr.append(tdRank, tdPlayer, tdSolved, tdTotal);
            tbody.appendChild(tr);
        });
    } catch (e) {
        tbody.innerHTML = '';
        empty.hidden = false;
        empty.textContent = UI[lang].packEmpty;
    }
}
