  try {

      const port = srv.address().port;
      const url = `http://localhost:${port}/api`;
      const fetch = (p, m, b) => fetch(`${url}${p}`, { method: m, headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(b) }).then(r => r.json());
      
      const s1 = await fetch('/session', 'POST', { case: k });
      assert(s1.budget > 0, 'budget');
      const suspects = Object.values(c.suspects);
      const innocent = suspects.find(s => s.id !== c.culprit);

      const a1 = await fetch('/accuse', 'POST', { sessionId: s1.sessionId, suspectId: innocent.id, lang: 'en' });
