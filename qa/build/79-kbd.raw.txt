// 7. Credits A11y
await runStep('credits-a11y', async () => {
  const aria = await evaluate(`document.getElementById('credits').getAttribute('aria-labelledby')`);
  if (aria !== 'credits-title') throw new Error(`aria-labelledby fail: ${aria}`);
  const title = await evaluate(`document.getElementById('credits-title').textContent`);
  if (title !== creditsData.title.en) throw new Error(`title fail: ${title}`);
  const listItems = await evaluate(`Array.from(document.querySelectorAll('#credits-list li')).map(li => li.textContent)`);
  for(let i=0; i<creditsData.backers.length; i++) {
    if (listItems[i] !== creditsData.backers[i]) throw new Error(`backer ${i} fail: ${listItems[i]}`);
  }
  const noteVisible = await evaluate(`!document.getElementById('credits-note').hidden`);
  if (!noteVisible) throw new Error('example note not visible');
});

// 8. Credits PT
await runStep('credits-pt', async () => {
  await evaluate(`document.getElementById('lpt').focus()`);
  await pressEnter();
  await waitFor(async () => (await evaluate(`document.getElementById('credits-title').textContent`)) === creditsData.title.pt);
  const note = await evaluate(`document.getElementById('credits-note').textContent`);
  if (note !== 'Nomes de exemplo') throw new Error(`note fail: ${note}`);
  const vText = await evaluate(`document.getElementById('verdict').innerText`);
  if (!vText.includes('Culpado')) throw new Error(`verdict pt fail: ${vText}`);
  const sCap = await evaluate(`document.getElementById('scores-caption').textContent`);
  if (sCap !== '10 melhores jogadores') throw new Error(`caption pt fail: ${sCap}`);
  const focus = await evaluate(`document.activeElement.id`);
  if (focus !== 'lpt') throw new Error(`focus lost on lang switch, got ${focus}`);
});
