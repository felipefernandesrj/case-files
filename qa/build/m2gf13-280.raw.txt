// locked question box (M2 game feel sprint 13): budget spent or game over looks locked, with a placeholder and aria-disabled
function renderAskLock() {
  const form = document.getElementById('ask-form');
  const q = document.getElementById('q');
  const ask = document.getElementById('ask-btn');
  if (!form || !q || !ask) return;
  const reason = gameOver ? 'over' : (budgetOut() ? 'out' : '');
  form.classList.toggle('is-locked', !!reason);
  form.dataset.lock = reason;
  q.placeholder = reason === 'over' ? LABELS[lang].overPlaceholder : (reason === 'out' ? LABELS[lang].outPlaceholder : '');
  for (const el of [q, ask]) {
    if (reason) el.setAttribute('aria-disabled', 'true');
    else el.removeAttribute('aria-disabled');
  }
}
