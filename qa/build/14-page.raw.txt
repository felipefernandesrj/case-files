<script type="module">
const API = '/api';
let lang = 'en';
let sessionId = null;
let caseData = null;
let selectedSuspect = null;
const histories = {}; // per suspectId [{q,a},...]

function setLang(newLang) {
  lang = newLang;
  document.documentElement.lang = newLang === 'pt' ? 'pt-BR' : 'en';
  document.getElementById('len').setAttribute('aria-pressed', newLang === 'en');
  document.getElementById('lpt').setAttribute('aria-pressed', newLang === 'pt');
  renderCase();
  renderAccuseButtons();
  renderConversation();
  document.getElementById('question-label').textContent = newLang === 'pt' ? 'Pergunta' : 'Question';
  document.getElementById('ask-btn').textContent = newLang === 'pt' ? 'Perguntar' : 'Ask';
  document.getElementById('player-label').textContent = newLang === 'pt' ? 'Seu nome' : 'Your name';
  document.getElementById('suspects-title').textContent = newLang === 'pt' ? 'Suspeitos' : 'Suspects';
  document.getElementById('conversation-title').textContent = newLang === 'pt' ? 'Conversa' : 'Conversation';
  document.getElementById('accuse-title').textContent = newLang === 'pt' ? 'Acusar' : 'Accuse';
  document.getElementById('title').textContent = newLang === 'pt' ? 'Mistério FullCup' : 'FullCup Mystery';
}

document.getElementById('len').addEventListener('click',()=>setLang('en'));
document.getElementById('lpt').addEventListener('click',()=>setLang('pt'));

async function init() {
  const sessRes = await fetch(API + '/session', {method:'POST'});
  const sessData = await sessRes.json();
  sessionId = sessData.sessionId;
  await loadCase();
  setLang('en');
}
async function loadCase() {
  const res = await fetch(API + '/case?lang=' + lang);
  caseData = await res.json();
}
function renderCase() {
  if (!caseData) return;
  document.getElementById('intro').innerHTML = `<p>${caseData.intro}</p>`;
  const btnContainer = document.getElementById('suspect-buttons');
  btnContainer.innerHTML = '';
  caseData.suspects.forEach(s => {
    const btn = document.createElement('button');
    btn.textContent = s.name;
    btn.dataset.id = s.id;
    btn.setAttribute('aria-pressed','false');
    btn.addEventListener('click',()=>selectSuspect(s.id));
    btnContainer.appendChild(btn);
  });
}
function selectSuspect(id) {
  selectedSuspect = id;
  document.querySelectorAll('#suspect-buttons button').forEach(b=>{
    b.setAttribute('aria-pressed', b.dataset.id===id ? 'true':'false');
  });
  document.getElementById('q').focus();
  renderConversation();
}
function renderConversation() {
  const log = document.getElementById('log');
  log.innerHTML = '';
  if (!selectedSuspect) return;
  const hist = histories[selectedSuspect] || [];
  hist.forEach(pair=>{
    const pQ = document.createElement('p');
    pQ.textContent = (lang==='pt'?'Pergunta: ':'Q: ') + pair.q;
    const pA = document.createElement('p');
    pA.textContent = (lang==='pt'?'Resposta: ':'A: ') + pair.a;
    log.appendChild(pQ);
    log.appendChild(pA);
  });
}
async function askQuestion(e){
  e.preventDefault();
  if (!selectedSuspect) return;
  const question = document.getElementById('q').value.trim();
  if (!question) return;
  document.getElementById('ask-btn').disabled = true;
  const log = document.getElementById('log');
  const thinking = document.createElement('p');
  thinking.textContent = '…thinking';
  log.appendChild(thinking);
  try{
    const res = await fetch(API + '/ask', {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({sessionId, suspectId:selectedSuspect, question, lang})
    });
    const data = await res.json();
    if (!res.ok) throw data;
    const answer = data.answer;
    // store
    if (!histories[selectedSuspect]) histories[selectedSuspect]=[];
    histories[selectedSuspect].push({q:question,a:answer});
    renderConversation();
  }catch(err){
    showError(err.message||'Error');
  }finally{
    document.getElementById('ask-btn').disabled = false;
    document.getElementById('q').value = '';
  }
}
async function accuse(suspectId){
  const playerName = document.getElementById('player').value.trim()||'anon';
  try{
    const res = await fetch(API + '/accuse', {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({sessionId, suspectId, player:playerName, lang})
    });
    const data = await res.json();
    if (!res.ok) throw data;
    const verdict = document.getElementById('verdict');
    verdict.hidden = false;
    verdict.innerHTML = `<p>${data.correct ? (lang==='pt'?'Acerto!':'Correct!') : (lang==='pt'?'Errou!':'Wrong!')}</p>
      <p>${lang==='pt'?'Culprit':'Culprit'}: ${data.culpritName}</p>
      <p>${lang==='pt'?'Solution':'Solution'}: ${data.solution}</p>
      <p>${lang==='pt'?'Score':'Score'}: ${data.score}</p>`;
    verdict.focus();
    await loadScores();
  }catch(err){
    showError(err.message||'Error');
  }
}
function renderAccuseButtons(){
  if (!caseData) return;
  const container = document.getElementById('accuse-buttons');
  container.innerHTML = '';
  caseData.suspects.forEach(s=>{
    const btn = document.createElement('button');
    btn.textContent = (lang==='pt'?'Acusar ':'Accuse ') + s.name;
    btn.addEventListener('click',()=>accuse(s.id));
    container.appendChild(btn);
  });
}
async function loadScores(){
  const sec = document.getElementById('scores-section');
  const tbody = document.getElementById('scores-body');
  const res = await fetch(API + '/scores');
  const data = await res.json();
  tbody.innerHTML = '';
  data.forEach((row,i)=>{
    const tr = document.createElement('tr');
    tr.innerHTML = `<td>${i+1}</td><td>${row.player}</td><td>${row.score}</td><td>${row.questions}</td>`;
    tbody.appendChild(tr);
  });
  sec.hidden = false;
}
function showError(msg){
  const el = document.getElementById('err');
  el.textContent = msg;
  el.hidden = false;
  setTimeout(()=>{el.hidden=true;},5000);
}
document.getElementById('ask-form').addEventListener('submit',askQuestion);
init();
</script>
</body>
</html>

import http from 'http';
import { readFileSync, promises as fs } from 'fs';
import { resolve, dirname, join } from 'path';
import { fileURLToPath } from 'url';
import { loadCase, publicCase, fairPlay, systemPrompt, capWords } from './case.js';
import { createProxy } from './proxy.js';
import { mockAnswer } from './mock.js';
import { friendly } from './messages.js';
import { createScoreboard } from './scoreboard.js';
import crypto from 'crypto';

const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

function readKey(optsKey) {
  if (optsKey) return optsKey.trim();
  const envKey = process.env.FULLCUP_KEY;
  if (envKey) return envKey.trim();
  try {
    const fileKey = readFileSync('.fullcup-key', 'utf8').trim();
    return fileKey || null;
  } catch {
    return null;
  }
}

export function createServer({
  caseFile = 'content/case.json',
  scoreFile = 'data/scores.json',
  key: optKey,
  baseUrl,
  fetchImpl,
  mock: optMock,
}) {
  const key = readKey(optKey);
  const mockMode = typeof optMock === 'boolean' ? optMock : (process.env.MOCK === '1' || !key);
  const caseObj = loadCase(caseFile);
  const fairness = fairPlay(caseObj);
  if (!fairness.ok) {
    console.error('Case fairness errors:', fairness.reasons);
    // continue anyway
  }

  const proxy = mockMode
    ? null
    : createProxy({ key, baseUrl: baseUrl || process.env.FULLCUP_BASE_URL || 'https://qa-api.fullcup.ai/v1', fetchImpl });

  const scoreboard = createScoreboard(scoreFile);

  const sessions = new Map(); // sessionId -> { histories: {suspectId: [{q,a}]}, questionCount, accused:false }

  const server = http.createServer(async (req, res) => {
    try {
      const url = new URL(req.url, `http://${req.headers.host}`);
      const method = req.method.toUpperCase();

      // Helper to send JSON
      const send = (code, obj) => {
        const payload = JSON.stringify(obj);
        res.writeHead(code, { 'Content-Type': 'application/json' });
        res.end(payload);
      };

      // Parse JSON body
      const parseBody = async () => {
        const chunks = [];
        for await (const chunk of req) chunks.push(chunk);
        const data = Buffer.concat(chunks).toString();
        return data ? JSON.parse(data) : {};
      };

      // Route handling
      if (method === 'GET' && url.pathname === '/') {
        const indexPath = join(__dirname, '..', 'public', 'index.html');
        const content = await fs.readFile(indexPath, 'utf8');
        res.writeHead(200, { 'Content-Type': 'text/html' });
        res.end(content);
        return;
      }

      if (method === 'GET' && url.pathname === '/api/status') {
        send(200, { mode: mockMode ? 'mock' : 'live', caseId: caseObj.id });
        return;
      }

      if (method === 'GET' && url.pathname === '/api/case') {
        const lang = url.searchParams.get('lang') || 'en';
        send(200, publicCase(caseObj, lang));
        return;
      }

      if (method === 'POST' && url.pathname === '/api/session') {
        const sessionId = crypto.randomUUID();
        sessions.set(sessionId, { histories: {}, questionCount: 0, accused: false });
        send(200, { sessionId });
        return;
      }

      // All other POST routes need JSON body
      if (method === 'POST' && (url.pathname === '/api/ask' || url.pathname === '/api/accuse')) {
        const body = await parseBody();

        const { sessionId, suspectId, lang = 'en' } = body;
        const session = sessions.get(sessionId);
        if (!session) {
          send(400, { error: 'bad_request', message: 'unknown session' });
          return;
        }

        const suspect = caseObj.suspects.find(s => s.id === suspectId);
        if (!suspect) {
          send(400, { error: 'bad_request', message: 'unknown suspect' });
          return;
        }

        if (url.pathname === '/api/ask') {
          const { question } = body;
          if (!question || typeof question !== 'string' || question.trim().length === 0) {
            send(400, { error: 'bad_request', message: 'empty question' });
            return;
          }
          if (question.length > 500) {
            send(400, { error: 'bad_request', message: 'question too long' });
            return;
          }

          // Build messages
          const prior = session.histories[suspectId] || [];
          const messages = [
            { role: 'system', content: systemPrompt(caseObj, suspectId, lang) },
            ...prior.flatMap(pair => [
              { role: 'user', content: pair.q },
              { role: 'assistant', content: pair.a },
            ]),
            { role: 'user', content: question },
          ];

          let answer;
          try {
            if (mockMode) {
              answer = mockAnswer(caseObj, suspectId, question, lang);
            } else {
              answer = await proxy.chat(messages);
            }
            answer = capWords(answer, 80);
          } catch (e) {
            // Map proxy errors
            const codeMap = {
              no_capacity: { status: 402, code: 'no_capacity' },
              rate_limited: { status: 429, code: 'rate_limited' },
              bad_key: { status: 503, code: 'bad_key' },
              upstream: { status: 502, code: 'upstream' },
            };
            const map = codeMap[e.code] || { status: 502, code: 'upstream' };
            send(map.status, { error: map.code, message: friendly(map.code, lang) });
            return;
          }

          // Store history
          if (!session.histories[suspectId]) session.histories[suspectId] = [];
          session.histories[suspectId].push({ q: question, a: answer });
          session.questionCount += 1;

          send(200, { answer });
          return;
        }

        // /api/accuse
        if (url.pathname === '/api/accuse') {
          if (session.accused) {
            send(409, { error: 'conflict', message: 'already accused' });
            return;
          }
          const { player = 'anon' } = body;
          const playerName = String(player).trim().slice(0, 24) || 'anon';
          const correct = suspectId === caseObj.culprit;
          const culpritObj = caseObj.suspects.find(s => s.id === caseObj.culprit);
          const solution = caseObj.solution[lang] || caseObj.solution.en;
          const score = scoreboard.scoreFor(correct, session.questionCount);
          scoreboard.add({
            player: playerName,
            score,
            questions: session.questionCount,
            correct,
            lang,
            at: new Date().toISOString(),
          });
          session.accused = true;
          send(200, {
            correct,
            culprit: caseObj.culprit,
            culpritName: culpritObj ? culpritObj.name : '',
            solution,
            score,
            questions: session.questionCount,
          });
          return;
        }
      }

      if (method === 'GET' && url.pathname === '/api/scores') {
        send(200, scoreboard.top(10));
        return;
      }

      // Not found
      send(404, { error: 'not_found', message: 'route not found' });
    } catch (err) {
      // Unexpected error
      const payload = JSON.stringify({ error: 'internal', message: err.message || 'internal error' });
      res.writeHead(500, { 'Content-Type': 'application/json' });
      res.end(payload);
    }
  });

  // If run directly, start listening
  if (import.meta.url === `file://${process.argv[1]}`) {
    const port = process.env.PORT || 3000;
    server.listen(port, () => {
      console.log(`Server listening at http://localhost:${port}`);
    });
  }

  return server;
}
