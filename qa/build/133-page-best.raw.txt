let questionsUsed = 0;
function renderQuestions() {
  document.getElementById('questions-used').textContent = questionsText(lang, questionsUsed);
}

function renderBest() {
  const b = readBest(store);
  document.getElementById('best-title').textContent = LABELS[lang].bestTitle;
  document.getElementById('reset-best').textContent = LABELS[lang].reset;
  document.getElementById('best-en').textContent = bestText('en', b.en);
  document.getElementById('best-pt').textContent = bestText('pt', b.pt);
}

document.getElementById('reset-best').addEventListener('click', () => {
  const btn = document.getElementById('reset-best');
  let status = '';
  if (confirm(LABELS[lang].confirm)) {
    resetBest(store);
    status = LABELS[lang].done;
  } else {
    status = LABELS[lang].kept;
  }
  renderBest();
  document.getElementById('best-status').textContent = status;
  btn.focus();
});

document.getElementById('ask-form').addEventListener('submit',askQuestion);
