function fnSource(name) {
  const startIdx = html.indexOf(`function ${name}(`);
  if (startIdx === -1) throw new Error(`Function ${name} not found`);
  let depth = 0;
  for (let i = html.indexOf('{', startIdx); i < html.length; i++) {
    if (html[i] === '{') depth++;
    else if (html[i] === '}') { depth--; if (depth === 0) return html.substring(startIdx, i + 1); }
  }
  throw new Error(`Unmatched braces in ${name}`);
}

class El {
  constructor(tag) {
    this.tagName = tag;
    this.className = '';
    this.textContent = '';
    this.children = [];
    this.attrs = {};
    this.parent = null;
    this.cls = new Set();
    this.classList = {
      add: c => this.cls.add(c),
      remove: c => this.cls.delete(c),
      contains: c => this.cls.has(c)
    };
  }
  append(...nodes) {
    for (const node of nodes) {
      node.parent = this;
      this.children.push(node);
    }
  }
  appendChild(n) {
    this.append(n);
    return n;
  }
  removeChild(n) {
    const i = this.children.indexOf(n);
    if (i >= 0) this.children.splice(i, 1);
    n.parent = null;
    return n;
  }
  remove() {
    if (this.parent) this.parent.removeChild(this);
  }
  get firstElementChild() {
    return this.children[0] || null;
  }
  set innerHTML(v) {
    this.children = [];
  }
  get innerHTML() {
    return '';
  }
  setAttribute(k, v) {
    this.attrs[k] = String(v);
  }
  getAttribute(k) {
    return this.attrs[k];
  }
  addEventListener() {}
  get text() {
    return this.children.map(c => c.text !== undefined ? c.text : c.textContent).join('') || this.textContent;
  }
}

function fakeTimers() {
  let nextId = 1;
  const timers = [];
  return {
    timers,
    set: (fn, ms) => {
      const entry = { id: nextId++, fn, ms, cleared: false };
      timers.push(entry);
      return entry.id;
    },
    clear: (id) => {
      const t = timers.find(x => x.id === id);
      if (t) t.cleared = true;
    }
  };
}

function hintApi(env) {
  const src = [fnSource('stopHints'), fnSource('showToast'), fnSource('startHints')].join('\n').replace(/\bgameOver\b/g, 'isOver()').replace(/\bcaseData\b/g, 'getCase()').replace(/\blang\b/g, 'getLang()');
  return new Function('document', 'setTimeout', 'clearTimeout', 'getLang', 'getCase', 'isOver', `let hintTimers = []; ${src} return { stopHints, showToast, startHints, timers: () => hintTimers };`)(env.doc, env.timers.set, env.timers.clear, env.getLang, env.getCase, env.isOver);
}

check('case files carry hints', async () => {
  const files = ['content/case.json', 'content/cases/night-train.json'];
  for (const f of files) {
    const c = await loadCase(f);
    if (!Array.isArray(c.hints) || c.hints.length !== 3) throw new Error(`${f} hints length error`);
    const ids = c.hints.map(h => h.id);
    if (new Set(ids).size !== 3) throw new Error(`${f} duplicate ids`);
    for (let i = 0; i < 3; i++) {
      if (typeof c.hints[i].after !== 'number' || c.hints[i].after <= 0) throw new Error(`${f} invalid after`);
      if (i > 0 && c.hints[i].after <= c.hints[i-1].after) throw new Error(`${f} non-increasing after`);
      if (!c.hints[i].text.en || !c.hints[i].text.pt) throw new Error(`${f} missing translations`);
    }
  }
});

check('publicCase localises hints', async () => {
  const files = ['content/case.json', 'content/cases/night-train.json'];
  for (const f of files) {
    const c = await loadCase(f);
    for (const l of ['en', 'pt']) {
      const pc = await publicCase(c, l);
      if (pc.hints.length !== 3) throw new Error(`Length mismatch for ${l}`);
      pc.hints.forEach((h, i) => {
        if (h.text !== c.hints[i].text[l]) throw new Error(`Text mismatch for ${l}`);
        if (h.after !== c.hints[i].after) throw new Error(`After mismatch for ${l}`);
      });
    }
  }
});
