import { createReadStream, readFileSync } from 'node:fs';
import { resolve, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';
import { randomUUID } from 'node:crypto';
import { loadCase, publicCase, systemPrompt, capWords } from './case.js';
import { createProxy } from './proxy.js';
import { mockAnswer } from './mock.js';
import { friendly } from './messages.js';

/**
 * Helper to read request body as JSON.
 * @param {import('node:http').IncomingMessage} req
 * @returns {Promise<any>}
 */
function readJSON(req) {
  return new Promise((resolve, reject) => {
    let data = '';
    req.on('data', chunk => (data += chunk));
    req.on('end', () => {
      try {
        resolve(JSON.parse(data || '{}'));
      } catch (e) {
        reject(e);
      }
    });
    req.on('error', reject);
  });
}

/**
 * Helper to send JSON response.
 * @param {import('node:http').ServerResponse} res
 * @param {number} status
 * @param {any} obj
 */
function sendJSON(res, status, obj) {
  const payload = JSON.stringify(obj);
  res.writeHead(status, {
    'Content-Type': 'application/json',
    'Content-Length': Buffer.byteLength(payload),
  });
  res.end(payload);
}

/**
 * Helper to send a static file.
 * @param {import('node/http').ServerResponse} res
 * @param {string} filePath
 */
function sendFile(res, filePath) {
  const stream = createReadStream(filePath);
  stream.on('error', () => {
    res.writeHead(404);
    res.end('Not found');
  });
  stream.pipe(res);
}

/**
 * Create the game server.
 * @param {Object} opts
 * @param {string} opts.caseFile
 * @param {string} [opts.scoreFile='data/scores.json']
 * @param {string} [opts.key]
 * @param {string} [opts.baseUrl]
 * @param {Function} [opts.fetchImpl]
 * @param {boolean} [opts.mock]
 * @returns {import('node:http').Server}
 */
export function createServer({
  caseFile,
  scoreFile = 'data/scores.json',
  key,
  baseUrl,
  fetchImpl,
  mock,
}) {
  // Load case (synchronous)
  const c = loadCase(caseFile);

  // Resolve key (synchronous)
  let finalKey = key;
  if (!finalKey) {
    const envKey = process.env.FULLCUP_KEY;
    if (envKey) finalKey = envKey.trim();
    else {
      try {
        const keyPath = resolve(process.cwd(), '.fullcup-key');
        const raw = readFileSync(keyPath, 'utf8');
        finalKey = raw.trim();
      } catch (_) {
        finalKey = '';
      }
    }
  }

  const modeMock =
    typeof mock === 'boolean'
      ? mock
      : process.env.MOCK === '1' || !finalKey;

  const base = baseUrl || process.env.FULLCUP_BASE_URL || 'https://qa-api.fullcup.ai/v1';

  const proxy = modeMock
    ? null
    : createProxy({
        key: finalKey,
        baseUrl: base,
        fetchImpl,
      });

  const scoreboard = createScoreboard(scoreFile);

  // Session storage
  const sessions = new Map(); // sessionId -> {history:Map(suspectId=>msg[]), questions:number, accused:boolean}

  // Load static index.html
  const __dirname = dirname(fileURLToPath(import.meta.url));
  const indexPath = resolve(__dirname, '..', 'public', 'index.html');

  const server = httpCreateServer(async (req, res) => {
    try {
      const url = new URL(req.url, `http://${req.headers.host}`);
      const pathname = url.pathname;
      const queryLang = url.searchParams.get('lang') === 'pt' ? 'pt' : 'en';

      const getLang = (bodyLang) => {
        if (bodyLang === 'pt') return 'pt';
        if (bodyLang === 'en') return 'en';
        return queryLang;
      };

      // Routes
      if (req.method === 'GET' && pathname === '/') {
        return sendFile(res, indexPath);
      }

      if (req.method === 'GET' && pathname === '/api/status') {
        return sendJSON(res, 200, { mode: modeMock ? 'mock' : 'live', caseId: c.id });
      }

      if (req.method === 'GET' && pathname === '/api/case') {
        const lang = queryLang;
        return sendJSON(res, 200, publicCase(c, lang));
      }

      if (req.method === 'POST' && pathname === '/api/session') {
        const sessionId = randomUUID();
        sessions.set(sessionId, {
          history: new Map(),
          questions: 0,
          accused: false,
        });
        return sendJSON(res, 200, { sessionId });
      }

      if (req.method === 'POST' && pathname === '/api/ask') {
        const body = await readJSON(req);
        const { sessionId, suspectId, question, lang: bodyLang } = body;
        const lang = getLang(bodyLang);

        // Validation
        if (!sessionId || !sessions.has(sessionId)) {
          return sendJSON(res, 400, { error: 'bad_request', message: 'unknown session' });
        }
        const suspect = c.suspects.find((s) => s.id === suspectId);
        if (!suspect) {
          return sendJSON(res, 400, { error: 'bad_request', message: 'unknown suspect' });
        }
        if (!question || typeof question !== 'string' || !question.trim()) {
          return sendJSON(res, 400, { error: 'bad_request', message: 'empty question' });
        }
        if (question.length > 500) {
          return sendJSON(res, 400, { error: 'bad_request', message: 'question too long' });
        }

        const sess = sessions.get(sessionId);
        sess.questions += 1;

        // Build messages
        const messages = [];
        messages.push({ role: 'system', content: systemPrompt(c, suspectId, lang) });

        const hist = sess.history.get(suspectId) || [];
        for (const msg of hist) {
          messages.push({ role: msg.role, content: msg.content });
        }
        messages.push({ role: 'user', content: question });

        let answer;
        try {
          if (modeMock) {
            answer = mockAnswer(c, suspectId, question, lang);
          } else {
            answer = await proxy.chat(messages);
          }
        } catch (e) {
          const code = e.code;
          let status;
          if (code === 'no_capacity') status = 402;
          else if (code === 'rate_limited') status = 429;
          else if (code === 'bad_key') status = 503;
          else if (code === 'upstream') status = 502;
          else status = 502;
          return sendJSON(res, status, { error: code || 'upstream', message: friendly(code, lang) });
        }

        const trimmed = capWords(answer, 80);
        const newHist = [...hist, { role: 'assistant', content: trimmed }];
        sess.history.set(suspectId, newHist);

        return sendJSON(res, 200, { answer: trimmed, questions: sess.questions });
      }

      if (req.method === 'POST' && pathname === '/api/accuse') {
        const body = await readJSON(req);
        const { sessionId, suspectId, player, lang: bodyLang } = body;
        const lang = getLang(bodyLang);
        if (!sessionId || !sessions.has(sessionId)) {
          return sendJSON(res, 400, { error: 'bad_request', message: 'unknown session' });
        }
        const sess = sessions.get(sessionId);
        if (sess.accused) {
          return sendJSON(res, 409, { error: 'bad_request', message: 'already accused' });
        }
        const suspect = c.suspects.find((s) => s.id === suspectId);
        if (!suspect) {
          return sendJSON(res, 400, { error: 'bad_request', message: 'unknown suspect' });
        }

        const correct = suspectId === c.culprit;
        const culpritObj = c.suspects.find((s) => s.id === c.culprit);
        const culpritName = culpritObj ? culpritObj.name : '';
        const solution = c.solution?.[lang] ?? '';
        const playerName = typeof player === 'string' && player.trim() ? player.trim().slice(0, 24) : 'anon';
        const sc = scoreFor(correct, sess.questions);
        scoreboard.add({
          player: playerName,
          score: sc,
          questions: sess.questions,
          correct,
          lang,
          caseId: c.id,
        }).catch(() => {});

        sess.accused = true;

        return sendJSON(res, 200, {
          correct,
          culprit: c.culprit,
          culpritName,
          solution,
          score: sc,
          questions: sess.questions,
        });
      }

      if (req.method === 'GET' && pathname === '/api/scores') {
        const top = scoreboard.top(10, c.id);
        return sendJSON(res, 200, top);
      }

      // Not found
      res.writeHead(404);
      res.end('Not found');
    } catch (err) {
      const status = err instanceof SyntaxError ? 400 : 500;
      res.writeHead(status, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ error: 'internal_error', message: err.message }));
    }
  });

  // If executed directly, start listening
  if (import.meta.url === `file://${process.argv[1]}`) {
    const port = Number(process.env.PORT) || 3000;
    server.listen(port, () => {
      console.log(`Server listening at http://localhost:${port}`);
    });
  }

  return server;
}

import { URL } from 'node:url';

/**
 * Create a FullCup proxy client.
 * @param {Object} opts
 * @param {string} opts.key
 * @param {string} opts.baseUrl
 * @param {Function} [opts.fetchImpl=fetch]
 * @param {string} [opts.model='@cf/meta/llama-4-scout-17b-16e-instruct']
 * @returns {{chat(messages: Array<Object>): Promise<string>}}
 */
export function createProxy({
  key,
  baseUrl,
  fetchImpl = fetch,
  model = '@cf/meta/llama-4-scout-17b-16e-instruct',
}) {
  const endpoint = new URL('/chat/completions', baseUrl).toString();

  async function chat(messages) {
    const payload = {
      model,
      messages,
      max_tokens: 300,
    };

    const response = await fetchImpl(endpoint, {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${key}`,
        'Content-Type': 'application/json',
        'X-FullCup-Workload': 'case-files:play',
      },
      body: JSON.stringify(payload),
    });

    if (response.ok) {
      const data = await response.json();
      const content = data?.choices?.[0]?.message?.content;
      return typeof content === 'string' ? content : '';
    }

    // error handling
    const err = new Error('FullCup request failed');
    const status = response.status;

    let bodyJson;
    try {
      // Try JSON first
      bodyJson = await response.json();
    } catch (_) {
      try {
        const txt = await response.text();
        bodyJson = JSON.parse(txt);
      } catch (_) {
        bodyJson = null;
      }
    }

    if (status === 401 || status === 403) {
      err.code = 'bad_key';
    } else if (status === 402) {
      err.code = 'no_capacity';
    } else if (status === 429) {
      const lower = (bodyJson?.error?.code ?? bodyJson?.error?.message ?? '').toString().toLowerCase();
      if (lower.includes('capacity:empty')) {
        err.code = 'no_capacity';
      } else {
        err.code = 'rate_limited';
      }
    } else if (status >= 500 && status < 600) {
      err.code = 'upstream';
    } else {
      err.code = 'upstream';
    }

    throw err;
  }

  return { chat };
}

/**
 * Generate a mock answer based on the case data.
 * If the question contains any keyword from a clue belonging to the suspect,
 * the answer will include the clue's text (in the requested language).
 * Also, if the question exactly matches the clue's "ask" field, the text is returned.
 * @param {Object} caseData
 * @param {string} suspectId
 * @param {string} question
 * @param {'en'|'pt'} lang
 * @returns {string}
 */
export function mockAnswer(caseData, suspectId, question, lang) {
  const lowerQ = question.toLowerCase();
  const clues = caseData.clues.filter(c => c.from === suspectId);
  for (const clue of clues) {
    // exact ask match (case‑insensitive)
    if (clue.ask && clue.ask.toLowerCase() === lowerQ) {
      return clue.text[lang] ?? '';
    }
    // keyword match
    if (Array.isArray(clue.keywords)) {
      for (const kw of clue.keywords) {
        if (kw && lowerQ.includes(kw.toLowerCase())) {
          return clue.text[lang] ?? '';
        }
      }
    }
  }
  // fallback generic answer
  return lang === 'pt' ? 'Não sei' : "I don't know";
}

import { readFileSync, writeFileSync, existsSync } from 'node:fs';
import { resolve } from 'node:path';

/**
 * Create a scoreboard.
 * @param {string|null} filePath
 * @returns {{add: Function, top: Function}}
 */
export function createScoreboard(filePath) {
  const fullPath = filePath ? resolve(filePath) : null;
  let scores = [];

  if (fullPath && existsSync(fullPath)) {
    try {
      const raw = readFileSync(fullPath, 'utf8');
      scores = JSON.parse(raw);
    } catch (_) {
      scores = [];
    }
  }

  async function add(entry) {
    scores.push({ ...entry, ts: Date.now() });
    if (fullPath) {
      try {
        writeFileSync(fullPath, JSON.stringify(scores, null, 2));
      } catch (_) {}
    }
  }

  function top(limit, caseId) {
    const filtered = scores.filter(s => s.caseId === caseId);
    const sorted = filtered.sort((a, b) => {
      if (b.score !== a.score) return b.score - a.score;
      return a.ts - b.ts; // earlier wins if same score
    });
    return sorted.slice(0, limit);
  }

  return { add, top };
}

/**
 * Compute score based on correctness and number of questions.
 * @param {boolean} correct
 * @param {number} questions
 * @returns {number}
 */
export function scoreFor(correct, questions) {
  if (!correct) return 0;
  const penalty = Math.max(0, questions - 2) * 5;
  const raw = 100 - penalty;
  return Math.max(10, raw);
}

          else if (code === 'upstream') status = 502;
          else status = 502;
          return sendJSON(res, status, { error: code || 'upstream', message: friendly(code, lang) });
        }

        const trimmed = capWords(answer, 80);
        const newHist = [...hist, { role: 'assistant', content: trimmed }];
        sess.history.set(suspectId, newHist);

        return sendJSON(res, 200, { answer: trimmed, questions: sess.questions });
      }

      if (req.method === 'POST' && pathname === '/api/accuse') {
        const body = await readJSON(req);
        const { sessionId, suspectId, player, lang: bodyLang } = body;
        const lang = getLang(bodyLang);
        if (!sessionId || !sessions.has(sessionId)) {
          return sendJSON(res, 400, { error: 'bad_request', message: 'unknown session' });
        }
        const sess = sessions.get(sessionId);
        if (sess.accused) {
          return sendJSON(res, 409, { error: 'bad_request', message: 'already accused' });
        }
        const suspect = c.suspects.find((s) => s.id === suspectId);
        if (!suspect) {
          return sendJSON(res, 400, { error: 'bad_request', message: 'unknown suspect' });
        }

        const correct = suspectId === c.culprit;
        const culpritObj = c.suspects.find((s) => s.id === c.culprit);
        const culpritName = culpritObj ? culpritObj.name : '';
        const solution = c.solution?.[lang] ?? '';
        const playerName = typeof player === 'string' && player.trim() ? player.trim().slice(0, 24) : 'anon';
        const sc = scoreFor(correct, sess.questions);
        scoreboard.add({
          player: playerName,
          score: sc,
          questions: sess.questions,
          correct,
          lang,
          caseId: c.id,
        }).catch(() => {});

        sess.accused = true;

        return sendJSON(res, 200, {
          correct,
          culprit: c.culprit,
          culpritName,
          solution,
          score: sc,
          questions: sess.questions,
        });
      }

      if (req.method === 'GET' && pathname === '/api/scores') {
        const top = scoreboard.top(10, c.id);
        return sendJSON(res, 200, top);
      }

      // Not found
      res.writeHead(404);
      res.end('Not found');
    } catch (err) {
      const status = err instanceof SyntaxError ? 400 : 500;
      res.writeHead(status, { 'Content-Type': 'application/json' });
      res.end(JSON.stringify({ error: 'internal_error', message: err.message }));
    }
  });

  // If executed directly, start listening
  if (import.meta.url === `file://${process.argv[1]}`) {
    const port = Number(process.env.PORT) || 3000;
    server.listen(port, () => {
      console.log(`Server listening at http://localhost:${port}`);
    });
  }

  return server;
}

  createServer({ caseFile: 'content/case.json' });

import { URL } from 'node:url';

/**
 * Create a FullCup proxy client.
 * @param {Object} opts
 * @param {string} opts.key
 * @param {string} opts.baseUrl
 * @param {Function} [opts.fetchImpl=fetch]
 * @param {string} [opts.model='@cf/meta/llama-4-scout-17b-16e-instruct']
 * @returns {{chat(messages: Array<Object>): Promise<string>}}
 */
export function createProxy({
  key,
  baseUrl,
  fetchImpl = fetch,
  model = '@cf/meta/llama-4-scout-17b-16e-instruct',
}) {
  const endpoint = new URL('/chat/completions', baseUrl).toString();

  async function chat(messages) {
    const payload = {
      model,
      messages,
      max_tokens: 300,
    };

    const response = await fetchImpl(endpoint, {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${key}`,
        'Content-Type': 'application/json',
        'X-FullCup-Workload': 'case-files:play',
      },
      body: JSON.stringify(payload),
    });

    if (response.ok) {
      const data = await response.json();
      const content = data?.choices?.[0]?.message?.content;
      return typeof content === 'string' ? content : '';
    }

    // error handling
    const err = new Error('FullCup request failed');
    const status = response.status;

    let bodyJson;
    try {
      // Try JSON first
      bodyJson = await response.json();
    } catch (_) {
      try {
        const txt = await response.text();
        bodyJson = JSON.parse(txt);
      } catch (_) {
        bodyJson = null;
      }
    }

    if (status === 401 || status === 403) {
      err.code = 'bad_key';
    } else if (status === 402) {
      err.code = 'no_capacity';
    } else if (status === 429) {
      const lower = (bodyJson?.error?.code ?? bodyJson?.error?.message ?? '').toString().toLowerCase();
      if (lower.includes('capacity:empty')) {
        err.code = 'no_capacity';
      } else {
        err.code = 'rate_limited';
      }
    } else if (status >= 500 && status < 600) {
      err.code = 'upstream';
    } else {
      err.code = 'upstream';
    }

    throw err;
  }

  return { chat };
}

/**
 * Generate a mock answer based on the case data.
 * If the question contains any keyword from a clue belonging to the suspect,
 * the answer will include the clue's text (in the requested language).
 * Also, if the question exactly matches the clue's "ask" field, the text is returned.
 * @param {Object} caseData
 * @param {string} suspectId
 * @param {string} question
 * @param {'en'|'pt'} lang
 * @returns {string}
 */
export function mockAnswer(caseData, suspectId, question, lang) {
  const lowerQ = question.toLowerCase();
  const clues = caseData.clues.filter(c => c.from === suspectId);
  for (const clue of clues) {
    // exact ask match (case‑insensitive)
    if (clue.ask && clue.ask.toLowerCase() === lowerQ) {
      return clue.text[lang] ?? '';
    }
    // keyword match
    if (Array.isArray(clue.keywords)) {
      for (const kw of clue.keywords) {
        if (kw && lowerQ.includes(kw.toLowerCase())) {
          return clue.text[lang] ?? '';
        }
      }
    }
  }
  // fallback generic answer
  return lang === 'pt' ? 'Não sei' : "I don't know";
}

import { readFileSync, writeFileSync, existsSync } from 'node:fs';
import { resolve } from 'node:path';

/**
 * Create a scoreboard.
 * @param {string|null} filePath
 * @returns {{add: Function, top: Function}}
 */
export function createScoreboard(filePath) {
  const fullPath = filePath ? resolve(filePath) : null;
  let scores = [];

  if (fullPath && existsSync(fullPath)) {
    try {
      const raw = readFileSync(fullPath, 'utf8');
      const data = JSON.parse(raw);
      scores = Array.isArray(data) ? data : [];
    } catch (_) {
      scores = [];
    }
  }

  async function add(entry) {
    scores.push({ ...entry, ts: Date.now() });
    if (fullPath) {
      try {
        writeFileSync(fullPath, JSON.stringify(scores, null, 2));
      } catch (_) {}
    }
  }

  function top(limit, caseId) {
    const filtered = scores.filter(s => s.caseId === caseId);
    const sorted = filtered.sort((a, b) => {
      if (b.score !== a.score) return b.score - a.score;
      return a.ts - b.ts;
    });
    return sorted.slice(0, limit);
  }

  return { add, top };
}

/**
 * Compute score based on correctness and number of questions.
 * @param {boolean} correct
 * @param {number} questions
 * @returns {number}
 */
export function scoreFor(correct, questions) {
  if (!correct) return 0;
  const penalty = Math.max(0, questions - 2) * 5;
  const raw = 100 - penalty;
  return Math.max(10, raw);
}

import fetch from 'node-fetch';
import readline from 'readline';

/**
 * Simple CLI.
 * Reads commands: ask <suspectId> <question>
 * accuse <...>
 */
async function main() {
  const port = process.env.PORT || 3000;
  const base = `http://localhost:${port}`;
  const rl = readline.createInterface({ input: process.stdin, output: process.stdout });
  rl.setPrompt('> ');
  rl.prompt();

  // start a session for CLI
  const sessionRes = await fetch(`${base}/api/session`);
  const sessionData = await sessionRes.json();
  const sessionId = sessionData.sessionId;

  for await (const line of rl) {
    const parts = line.trim().split(/\s+/);
    const cmd = parts[0]?.toLowerCase();
    if (!cmd) continue;

    if (cmd === 'ask' || cmd === 'pergunta') {
      const suspectId = parts[1];
      const question = parts.slice(2).join(' ');
      const res = await fetch(`${base}/api/ask`, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ sessionId, suspectId, question, lang: cmd === 'pergunta' ? 'pt' : 'en' }),
      });
      const data = await res.json();
      console.log(data.answer || data.error);
    } else if (cmd === 'accuse' || cmd === 'acusar') {
      const suspectId = parts[1];
      const res = await fetch(`${base}/api/accuse`, {
        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ sessionId, suspectId, player: 'cli', lang: cmd === 'acusar' ? 'pt' : 'en' }),
      });
      const data = await res.json();
      console.log(data.solution || data.message);
    } else if (cmd === 'exit' || cmd === 'quit') {
      break;
    }
    rl.prompt();
  }
  rl.close();
}
main();
