/**
 * @param {'en'|'pt'} lang - Language of the record
 * @param {{score: number, questions: number, at: string}|null} entry - The record entry
 * @param {'en'|'pt'} [ui=lang] - UI language for the wording
 * @returns {string}
 */
export function bestText(lang, entry, ui = lang) {
  const names = { en: 'English', pt: 'português' };
  const scores = { en: 'points', pt: 'pontos' };
  const noEntry = { en: `No best score yet in ${names.en}.`, pt: `Ainda sem recorde em ${names.pt}.` };
  const best = { en: `Your best in ${names.en}:`, pt: `Seu recorde em ${names.pt}:` };

  if (!entry) return noEntry[ui];

  const q = entry.questions === 1 
    ? (ui === 'pt' ? "1 pergunta" : "1 question") 
    : (ui === 'pt' ? `${entry.questions} perguntas` : `${entry.questions} questions`);

  return `${best[ui]} ${entry.score} ${scores[ui]} com ${q}.`.replace(' com ', ' with ');
}
