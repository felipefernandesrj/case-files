let pins = [];
const visitedSteps = new Set(['briefing']);
const STEP_ORDER = ['briefing', 'locations', 'suspects', 'talk', 'board', 'accuse'];
function renderProgress() {
  const list = document.getElementById('progress-list');
  const countEl = document.getElementById('progress-count');
  if (!list) return;
  list.textContent = '';
  STEP_ORDER.forEach(step => {
    const el = document.querySelector(`[data-step-label="${step}"]`);
    if (!el) return;
    const li = document.createElement('li');
    const isVisited = visitedSteps.has(step);
    const isCurrent = document.body.dataset.step === step;
    li.className = isVisited ? 'done' : 'todo';
    li.setAttribute('data-done', isVisited);
    if (isCurrent) li.setAttribute('aria-current', 'step');
    const labelText = el.textContent;
    li.textContent = (isVisited ? '✓ ' : '○ ') + labelText;
    list.appendChild(li);
  });
  if (countEl) countEl.textContent = `${visitedSteps.size}/6`;
}
