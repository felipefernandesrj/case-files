function fnSource(name) {
  const startIdx = html.indexOf(`function ${name}(`);
  if (startIdx === -1) throw new Error(`Function ${name} not found`);
  let depth = 0;
  for (let i = html.indexOf('{', startIdx); i < html.length; i++) {
    if (html[i] === '{') depth++;
    else if (html[i] === '}') { depth--; if (depth === 0) return html.substring(startIdx, i + 1); }
  }
  throw new Error(`Unmatched braces in ${name}`);
}
const results = [];

class El {
  constructor(id) {
    this.id = id;
    this.className = '';
    this.textContent = '';
    this.hidden = false;
    this.offsetWidth = 0;
    this.cls = new Set();
    this.classList = {
      add: c => this.cls.add(c),
      remove: c => this.cls.delete(c),
      contains: c => this.cls.has(c)
    };
  }
  focus() { El.focused = this.id; }
}
El.focused = null;

function fakeIntervals() {
  let idCounter = 0;
  const list = [];
  const set = (fn, ms) => {
    const id = ++idCounter;
    const entry = { id, fn, ms, cleared: false };
    list.push(entry);
    return id;
  };
  const clear = (id) => {
    const entry = list.find(e => e.id === id);
    if (entry) entry.cleared = true;
  };
  return { list, set, clear };
}

function contApi(opts) {
  const ids = ['continue', 'continue-title', 'continue-num', 'continue-note', 'continue-yes', 'continue-no', 'continue-close', 'continue-live', 'restart-btn'];
  const els = {};
  ids.forEach(i => { els[i] = new El(i); });
  const doc = { getElementById: i => els[i] || null };
  const iv = fakeIntervals();
  const calls = { verdict: [], restart: [] };
  El.focused = null;
  const from = html.match(/const CONT_FROM = (\d+);/);
  if (!from) throw new Error('CONT_FROM not found');
  const src = ['contLabels', 'renderContinue', 'contSay', 'startContinue', 'endContinue', 'stopContinue', 'closeContinue', 'continueGame'].map(fnSource).join('\n').replace(/\blastAccuseResult\b/g, 'getResult()').replace(/\blang\b/g, 'getLang()');
  const api = new Function('document', 'setInterval', 'clearInterval', 'getLang', 'getResult', 'renderVerdict', 'restartGame', `const CONT_FROM = ${from[1]}; let contTimer = null; let contLeft = 0; let contState = 'off'; ${src} return { renderContinue, startContinue, endContinue, stopContinue, closeContinue, continueGame, get state() { return contState; }, get left() { return contLeft; } };`)(doc, iv.set, iv.clear, () => opts.lang, () => opts.result, (m) => calls.verdict.push(m), (s) => calls.restart.push(s));
  return { api, els, iv, calls };
}
