import http from 'http';
import express from 'express';
import { json } from 'body-parser';
import { loadCase } from './case.js';
import { createProxy } from './proxy.js';
import { friendly } from './messages.js';
import { createScoreboard } from './scoreboard.js';
import { systemPrompt } from './systemPrompt.js';
import { v4 as uuidv4 } from 'uuid';
import { readFileSync } from 'fs';
import path from 'path';

/**
 * Create and configure the HTTP server.
 *
 * Options:
 *  - caseFile: path to the case JSON file (required)
 *  - mock: boolean indicating mock mode (default false)
 *  - key: FullCup API key (optional, read from env or file if omitted)
 *  - baseUrl: FullCup base URL (optional, defaults to env FULLCUP_BASE_URL)
 *  - fetchImpl: custom fetch implementation (for testing)
 *  - scoreFile: path to scores JSON file (optional, null disables persistence)
 */
export function createServer({
  caseFile,
  mock = false,
  key,
  baseUrl,
  fetchImpl,
  scoreFile = path.join(process.cwd(), 'scores.json'),
}) {
  const app = express();
  app.use(json());

  // Load case data once at startup
  const caseData = loadCase(caseFile);

  // Initialise proxy (real or mock)
  const proxy = createProxy({
    key: key ?? getKey(),
    baseUrl: baseUrl ?? process.env.FULLCUP_BASE_URL ?? 'https://qa-api.fullcup.ai/v1',
    fetchImpl,
    mock,
    caseData,
  });

  // Initialise scoreboard (may be disabled with null)
  const scoreboard = createScoreboard(scoreFile);

  // ---------- Routes ----------
  app.get('/api/status', (req, res) => {
    res.json({ mode: mock ? 'mock' : 'real', caseId: caseData.id });
  });

  app.get('/api/case', (req, res) => {
    const lang = req.query.lang === 'pt' ? 'pt' : 'en';
    const publicCase = {
      id: caseData.id,
      title: caseData.title[lang],
      intro: caseData.intro[lang],
      suspects: caseData.suspects.map(s => ({
        id: s.id,
        role: s.role[lang],
        persona: s.persona[lang],
      })),
    };
    res.json(publicCase);
  });

  // Simple in‑memory session store
  const sessions = new Map();

  app.post('/api/session', (req, res) => {
    const sessionId = uuidv4();
    sessions.set(sessionId, { questions: 0, accused: false });
    res.json({ sessionId });
  });

  app.post('/api/ask', async (req, res) => {
    const { sessionId, suspectId, question, lang } = req.body;
    if (!sessionId || !sessions.has(sessionId)) {
      return res.status(400).json({ error: 'invalid session' });
    }
    if (!question || typeof question !== 'string' || question.trim() === '' || question.length > 500) {
      return res.status(400).json({ error: 'invalid question' });
    }
    const suspect = caseData.suspects.find(s => s.id === suspectId);
    if (!suspect) {
      return res.status(400).json({ error: 'unknown suspect' });
    }

    const session = sessions.get(sessionId);
    session.questions += 1;

    try {
      const answer = await proxy.chat([
        { role: 'system', content: systemPrompt(caseData, suspectId, lang) },
        { role: 'user', content: question },
      ]);
      // Enforce max 80 words in mock mode (real mode handled by proxy)
      const wordCount = answer.split(/\s+/).filter(Boolean).length;
      if (mock && wordCount > 80) {
        return res.status(400).json({ error: 'answer exceeds 80 words' });
      }
      res.json({ answer, questions: session.questions });
    } catch (e) {
      // Map proxy errors to HTTP status + friendly messages
      const code = e.code || 'upstream';
      const statusMap = {
        bad_key: 401,
        no_capacity: 402,
        rate_limited: 429,
        upstream: 502,
      };
      const status = statusMap[code] || 500;
      const message = friendly(code, lang);
      res.status(status).json({ error: code, message });
    }
  });

  app.post('/api/accuse', (req, res) => {
    const { sessionId, suspectId, player, lang } = req.body;
    if (!sessionId || !sessions.has(sessionId)) {
      return res.status(400).json({ error: 'invalid session' });
    }
    const sess = sessions.get(sessionId);
    if (sess.accused) {
      return res.status(409).json({ error: 'already accused' });
    }
    sess.accused = true;
    const correct = suspectId === caseData.culprit;
    const score = correct ? scoreFor(true, sess.questions) : 0;
    if (scoreboard && player) {
      scoreboard.record({ player, score, correct, timestamp: Date.now() });
    }
    res.json({ correct, score });
  });

  app.get('/api/scores', (req, res) => {
    if (!scoreboard) return res.json([]);
    const list = scoreboard.list();
    res.json(list);
  });

  // Serve static UI
  const uiPath = path.join(import.meta.url.startsWith('file:') ? path.dirname(fileURLToPath(import.meta.url)) : process.cwd(), '..', 'public');
  app.use(express.static(uiPath));

  // Fallback to index.html for SPA routes
  app.get('*', (req, res) => {
    res.sendFile(path.join(uiPath, 'index.html'));
  });

  const server = http.createServer(app);

  // -----------------------------------------------------------------
  // Automatically start listening on a random port (or PORT env if set)
  // This change ensures tests that await the 'listening' event work
  // without requiring callers to invoke server.listen manually.
  // -----------------------------------------------------------------
  const listenPort = process.env.PORT ? parseInt(process.env.PORT, 10) : 0;
  server.listen(listenPort);

  return server;
}

/**
 * Retrieve the FullCup API key from environment or .fullcup-key file.
 */
function getKey() {
  if (process.env.FULLCUP_KEY) return process.env.FULLCUP_KEY.trim();
  try {
    const filePath = path.resolve('.fullcup-key');
    return readFileSync(filePath, 'utf8').trim() || null;
  } catch (_) {
    return null;
  }
}

/**
 * Scoring formula as described in the specification.
 */
function scoreFor(correct, questions) {
  if (!correct) return 0;
  const base = 100;
  const penalty = 5 * Math.max(0, questions - 2);
  return Math.max(10, base - penalty);
}
