function fnSource(name) {
  const startIdx = html.indexOf(`function ${name}(`);
  if (startIdx === -1) throw new Error(`Function ${name} not found`);
  let depth = 0;
  for (let i = html.indexOf('{', startIdx); i < html.length; i++) {
    if (html[i] === '{') depth++;
    else if (html[i] === '}') { depth--; if (depth === 0) return html.substring(startIdx, i + 1); }
  }
  throw new Error(`Unmatched braces in ${name}`);
}

function png(file) {
  const b = fs.readFileSync(file);
  must(b.readUInt32BE(0) === 0x89504e47, 'not a png: ' + file);
  let w = 0, h = 0, type = 0, depth = 0; const idat = [];
  for (let o = 8; o < b.length;) {
    const len = b.readUInt32BE(o); const t = b.toString('ascii', o + 4, o + 8); const d = b.subarray(o + 8, o + 8 + len);
    if (t === 'IHDR') { w = d.readUInt32BE(0); h = d.readUInt32BE(4); depth = d[8]; type = d[9]; }
    if (t === 'IDAT') idat.push(d);
    o += 12 + len;
  }
  must(depth === 8 && (type === 6 || type === 2), `unsupported png ${file} depth ${depth} type ${type}`);
  const bpp = type === 6 ? 4 : 3; const raw = zlib.inflateSync(Buffer.concat(idat)); const stride = w * bpp;
  const px = Buffer.alloc(h * stride);
  for (let y = 0; y < h; y++) {
    const f = raw[y * (stride + 1)];
    for (let x = 0; x < stride; x++) {
      const v = raw[y * (stride + 1) + 1 + x]; const a = x >= bpp ? px[y * stride + x - bpp] : 0; const up = y ? px[(y - 1) * stride + x] : 0; const c = y && x >= bpp ? px[(y - 1) * stride + x - bpp] : 0;
      let r = v;
      if (f === 1) r = v + a; else if (f === 2) r = v + up; else if (f === 3) r = v + ((a + up) >> 1);
      else if (f === 4) { const p = a + up - c; const pa = Math.abs(p - a), pb = Math.abs(p - up), pc = Math.abs(p - c); r = v + (pa <= pb && pa <= pc ? a : pb <= pc ? up : c); }
      px[y * stride + x] = r & 255;
    }
  }
  return { w, h, bpp, at: (x, y) => [...px.subarray(y * stride + x * bpp, y * stride + x * bpp + bpp)].concat(bpp === 3 ? [255] : []) };
}
