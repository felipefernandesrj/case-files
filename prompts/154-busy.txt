async function askQuestion(e){
  e.preventDefault();
  if (!selectedSuspect) return;
  const question = document.getElementById('q').value.trim();
  if (!question) return;
  const qInput = document.getElementById('q');
  const askBtn = document.getElementById('ask-btn');
  const askForm = document.getElementById('ask-form');
  const accuseBtns = document.querySelectorAll('#accuse-buttons button');
  askBtn.disabled = true;
  qInput.disabled = true;
  accuseBtns.forEach(b => { b.disabled = true; });
  askForm.setAttribute('aria-busy', 'true');
  const log = document.getElementById('log');
  const thinking = document.createElement('p');
  thinking.textContent = LABELS[lang].thinking;
  log.appendChild(thinking);
  try{
    const res = await fetch(API + '/ask', {
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({sessionId, suspectId:selectedSuspect, question, lang})
    });
    const data = await res.json();
    if (!res.ok) throw data;
    const answer = data.answer;
    questionsUsed = data.questions;
    renderQuestions();
    if (!histories[selectedSuspect]) histories[selectedSuspect]=[];
    histories[selectedSuspect].push({q:question,a:answer});
    renderConversation();
  }catch(err){
    showError(err.message||'Error');
  }finally{
    if (thinking.parentNode) thinking.remove();
    askBtn.disabled = false;
    qInput.disabled = false;
    accuseBtns.forEach(b => { b.disabled = false; });
    askForm.removeAttribute('aria-busy');
    qInput.value = '';
    qInput.focus();
  }
}
